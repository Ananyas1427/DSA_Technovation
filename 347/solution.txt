class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
    unordered_map<int, int> mp;
    for (int x : nums) {
            mp[x]++;
        }
    vector<int>ans;
    for (auto x:mp)
    ans.push_back(x.first);
    sort(ans.begin(),ans.end(),[&](int a, int b){
        return mp[a]>mp[b];

    });
    vector< int >arr;

    for(int i = 0; i < k; i++) {
            arr.push_back(ans[i]);
        }
    return arr;

    }
};
